cmake_minimum_required(VERSION 3.16)
project(paCker VERSION 0.4.1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(PACKER_OUTPUT_DIRECTORY "" CACHE PATH "Flat artifact directory for build scripts")
if(PACKER_OUTPUT_DIRECTORY)
  foreach(kind RUNTIME LIBRARY ARCHIVE)
    set(CMAKE_${kind}_OUTPUT_DIRECTORY "${PACKER_OUTPUT_DIRECTORY}")
    foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
      set(CMAKE_${kind}_OUTPUT_DIRECTORY_${config} "${PACKER_OUTPUT_DIRECTORY}")
    endforeach()
  endforeach()
endif()
add_library(oblique STATIC src/oblique.c)
target_include_directories(oblique PUBLIC include)
add_library(oblique_shared SHARED src/oblique.c)
target_include_directories(oblique_shared PUBLIC include)
set_target_properties(oblique_shared PROPERTIES OUTPUT_NAME oblique WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(WIN32)
  set_target_properties(oblique PROPERTIES OUTPUT_NAME oblique_static)
endif()
if(NOT WIN32)
  target_link_libraries(oblique PUBLIC m)
  target_link_libraries(oblique_shared PUBLIC m)
endif()
add_executable(test_core tests/test_core.c)
target_link_libraries(test_core PRIVATE oblique)
enable_testing()
add_test(NAME core COMMAND test_core)
add_library(packer3d_geometry STATIC src/geometry3d.c)
target_include_directories(packer3d_geometry PUBLIC include)
if(NOT WIN32)
  target_link_libraries(packer3d_geometry PUBLIC m)
endif()
add_executable(test_geometry3d tests/test_geometry3d.c)
target_link_libraries(test_geometry3d PRIVATE packer3d_geometry)
add_test(NAME geometry3d COMMAND test_geometry3d)
set_target_properties(packer3d_geometry PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(packer3d STATIC src/packer3d.c src/search3d.c src/geometry3d.c src/oblique.c)
target_include_directories(packer3d PUBLIC include)
if(NOT WIN32)
  target_link_libraries(packer3d PUBLIC m)
endif()
if(WIN32)
  # MSVC names both a static library and a DLL import library *.lib.
  set_target_properties(packer3d PROPERTIES OUTPUT_NAME packer3d_static)
endif()
add_library(packer3d_shared SHARED src/packer3d.c src/search3d.c src/geometry3d.c src/oblique.c)
target_include_directories(packer3d_shared PUBLIC include)
set_target_properties(packer3d_shared PROPERTIES OUTPUT_NAME packer3d WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(NOT WIN32)
  target_link_libraries(packer3d_shared PUBLIC m)
endif()
add_executable(test_free3d tests/test_free3d.c)
target_link_libraries(test_free3d PRIVATE packer3d)
add_test(NAME free3d COMMAND test_free3d)
add_executable(test_search3d_refinement tests/search3d_refinement.c)
target_link_libraries(test_search3d_refinement PRIVATE packer3d_geometry)
add_test(NAME search3d_refinement COMMAND test_search3d_refinement)
add_executable(test_json3d tests/test_json3d.c src/json3d.c third_party/cJSON.c)
target_link_libraries(test_json3d PRIVATE packer3d)
add_test(NAME json3d COMMAND test_json3d)
add_executable(free3d_example examples/free3d.c)
target_link_libraries(free3d_example PRIVATE packer3d)
add_test(NAME free3d_example COMMAND free3d_example)
add_executable(ordinary_example examples/ordinary.c)
target_link_libraries(ordinary_example PRIVATE oblique)
add_test(NAME ordinary_example COMMAND ordinary_example)
include(CheckLanguage)
check_language(CXX)
if(CMAKE_CXX_COMPILER)
  enable_language(CXX)
  add_executable(cpp_link tests/cpp_link.cpp)
  set_target_properties(cpp_link PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON)
  target_link_libraries(cpp_link PRIVATE oblique packer3d)
  add_test(NAME cpp_link COMMAND cpp_link)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/cli.c")
  add_executable(packer src/cli.c src/json3d.c third_party/cJSON.c)
  target_include_directories(packer PRIVATE third_party)
  target_link_libraries(packer PRIVATE oblique packer3d)
  if(WIN32)
    target_link_libraries(packer PRIVATE shell32)
  endif()
endif()
include(GNUInstallDirs)
install(TARGETS oblique oblique_shared packer3d packer3d_shared ordinary_example free3d_example packer
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/oblique.h include/packer3d.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES examples/ordinary.c examples/free3d.c DESTINATION ${CMAKE_INSTALL_DATADIR}/packer/examples)
install(FILES LICENSE third_party/cJSON.LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/packer/licenses)
